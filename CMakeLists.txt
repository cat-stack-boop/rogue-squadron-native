cmake_minimum_required(VERSION 3.20)
project(RogueSquadronNative LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
file(GLOB GAME_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/build/candidate-generated/*.c")
if(NOT GAME_SOURCES)
  message(FATAL_ERROR "Run tools/analyze_code.py and N64Recomp build/candidate.toml first")
endif()
add_library(rogue_game STATIC ${GAME_SOURCES})
target_include_directories(rogue_game PRIVATE vendor/N64Recomp/include build/candidate-generated)
target_compile_options(rogue_game PRIVATE -fno-strict-aliasing -fwrapv -Werror=implicit-function-declaration
  -include "${CMAKE_CURRENT_SOURCE_DIR}/src/recomp_compat.h")
target_compile_definitions(rogue_game PRIVATE RS_CHECK_MEMORY)
add_subdirectory(vendor/N64ModernRuntime/ultramodern)
find_package(SDL2 CONFIG REQUIRED HINTS /opt/homebrew/lib/cmake/SDL2)
add_library(rogue_musyx STATIC src/native_rsp.c vendor/rsp-hle/src/musyx.c
  vendor/rsp-hle/src/audio.c vendor/rsp-hle/src/memory.c)
target_include_directories(rogue_musyx PRIVATE src vendor/rsp-hle/src)
target_compile_options(rogue_musyx PRIVATE -fwrapv -fno-strict-aliasing)
target_compile_definitions(rogue_musyx PRIVATE ROGUE_NATIVE_GUARDS)
set(ROGUE_RUNTIME_SOURCES src/native_video.cpp src/native_audio.cpp src/native_audio_device.cpp src/native_paths.cpp src/native_gamepad.cpp src/native_eeprom.cpp
  vendor/N64ModernRuntime/librecomp/src/ultra_translation.cpp
  vendor/N64ModernRuntime/librecomp/src/cont.cpp
  vendor/N64ModernRuntime/librecomp/src/ai.cpp)
add_executable(rogue_startup src/headless_runtime.cpp ${ROGUE_RUNTIME_SOURCES})
add_executable(rogue_result_fixture src/headless_runtime.cpp src/result_fixture.cpp ${ROGUE_RUNTIME_SOURCES})
target_compile_definitions(rogue_result_fixture PRIVATE ROGUE_RESULT_FIXTURE=1)
foreach(runtime_target IN ITEMS rogue_startup rogue_result_fixture)
target_compile_features(${runtime_target} PRIVATE cxx_std_20)
target_include_directories(${runtime_target} PRIVATE vendor/N64Recomp/include build/candidate-generated build vendor/GLideN64/src/inc)
target_include_directories(${runtime_target} PRIVATE vendor/N64ModernRuntime/librecomp/include/librecomp)
target_link_libraries(${runtime_target} PRIVATE rogue_game ultramodern rogue_musyx SDL2::SDL2 "-framework OpenGL")
target_link_options(${runtime_target} PRIVATE "-Wl,-export_dynamic")
target_compile_definitions(${runtime_target} PRIVATE ROGUE_PROJECT_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
endforeach()
add_executable(rogue_thread_probe src/thread_probe.cpp)
target_compile_features(rogue_thread_probe PRIVATE cxx_std_20)
target_link_libraries(rogue_thread_probe PRIVATE ultramodern)
add_executable(rogue_io_probe src/native_io_probe.cpp)
target_compile_features(rogue_io_probe PRIVATE cxx_std_20)
add_executable(rogue_rsp_probe src/rsp_probe.c)
target_link_libraries(rogue_rsp_probe PRIVATE rogue_musyx)
add_executable(rogue_audio_recovery_probe src/audio_recovery_probe.cpp src/native_audio.cpp src/native_audio_device.cpp)
target_compile_features(rogue_audio_recovery_probe PRIVATE cxx_std_20)
target_include_directories(rogue_audio_recovery_probe PRIVATE vendor/N64ModernRuntime/ultramodern/include)
target_link_libraries(rogue_audio_recovery_probe PRIVATE SDL2::SDL2)
add_executable(rogue_audio_device_probe src/audio_device_probe.cpp src/native_audio.cpp)
target_compile_features(rogue_audio_device_probe PRIVATE cxx_std_20)
target_include_directories(rogue_audio_device_probe PRIVATE vendor/N64ModernRuntime/ultramodern/include)
target_link_libraries(rogue_audio_device_probe PRIVATE SDL2::SDL2)
add_executable(rogue_boot_probe src/boot_state_probe.c)
target_include_directories(rogue_boot_probe PRIVATE src vendor/N64Recomp/include build/candidate-generated)
target_compile_options(rogue_boot_probe PRIVATE -fno-strict-aliasing -fwrapv)
target_link_libraries(rogue_boot_probe PRIVATE rogue_game)
add_executable(rogue_orientation_probe src/orientation_probe.c)
target_include_directories(rogue_orientation_probe PRIVATE src vendor/N64Recomp/include build/candidate-generated)
target_compile_options(rogue_orientation_probe PRIVATE -fno-strict-aliasing -fwrapv)
target_link_libraries(rogue_orientation_probe PRIVATE rogue_game)
add_executable(rogue_controller_probe src/controller_probe.c)
target_include_directories(rogue_controller_probe PRIVATE src vendor/N64Recomp/include build/candidate-generated)
target_compile_options(rogue_controller_probe PRIVATE -fno-strict-aliasing -fwrapv)
target_link_libraries(rogue_controller_probe PRIVATE rogue_game)
add_executable(rogue_test_controller_probe src/test_controller_probe.cpp)
target_compile_features(rogue_test_controller_probe PRIVATE cxx_std_20)
add_executable(rogue_entities_probe src/entities_probe.cpp)
target_compile_features(rogue_entities_probe PRIVATE cxx_std_20)
add_executable(rogue_gamepad_probe src/gamepad_probe.cpp src/native_gamepad.cpp)
target_compile_features(rogue_gamepad_probe PRIVATE cxx_std_20)
target_link_libraries(rogue_gamepad_probe PRIVATE SDL2::SDL2)
add_executable(rogue_eeprom_probe src/eeprom_probe.cpp src/native_eeprom.cpp)
target_compile_features(rogue_eeprom_probe PRIVATE cxx_std_20)
add_executable(rogue_save_codec_probe src/save_codec_probe.cpp src/native_eeprom.cpp)
target_compile_features(rogue_save_codec_probe PRIVATE cxx_std_20)
target_include_directories(rogue_save_codec_probe PRIVATE vendor/N64Recomp/include build/candidate-generated)
target_compile_options(rogue_save_codec_probe PRIVATE -fno-strict-aliasing -fwrapv)
target_link_libraries(rogue_save_codec_probe PRIVATE rogue_game "-lz")
